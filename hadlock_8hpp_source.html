88 template <typename Node, typename WeightFunc>
89 auto build_dual(const std::vector<std::vector<Node>>& faces,
90 WeightFunc weight) -> std::vector<std::vector<DualEdge<Node>>> {
91 const auto n_face = faces.size();
92
93 std::unordered_map<std::pair<Node, Node>, std::vector<int>> edge_face_map;
94 edge_face_map.reserve(n_face * 2);
95 for (size_t fi = 0; fi < n_face; ++fi) {
96 const auto& f = faces[fi];
97 const auto sz = f.size();
98 for (size_t i = 0; i < sz; ++i) {
99 auto u = f[i];
100 auto v = f[(i + 1) % sz];
101 if (u > v) std::swap(u, v);
102 edge_face_map[{u, v}].push_back(static_cast<int>(fi));
103 }
104 }
105
106 std::unordered_map<std::pair<int, int>, std::pair<int, std::pair<Node, Node>>> best;
107 best.reserve(edge_face_map.size());
108 for (const auto& [primal_key, face_ids] : edge_face_map) {
109 if (face_ids.size() < 2) continue;
110 const auto& [u, v] = primal_key;
111 const int w = weight(u, v);
112 for (size_t a = 0; a < face_ids.size(); ++a) {
113 for (size_t b = a + 1; b < face_ids.size(); ++b) {
114 int fi = face_ids[a];
115 int fj = face_ids[b];
116 if (fi == fj) continue; // bridge: both traversals hit the same face
117 if (fi > fj) std::swap(fi, fj);
118 auto it = best.find({fi, fj});
119 if (it == best.end() || w < it->second.first) {
120 best[{fi, fj}] = {w, {u, v}};
121 }
122 }
123 }
124 }
125
126 std::vector<std::vector<DualEdge<Node>>> dual(n_face);
127 auto* dual_raw = dual.data();
128 for (const auto& [key, info] : best) {
129 const auto& [fi, fj] = key;
130 const auto& [w, primal] = info;
131 dual_raw[fi].push_back({fj, w, primal});
132 dual_raw[fj].push_back({fi, w, primal});
133 }
134 return dual;
135 }

144 inline auto reconstruct_path(const std::vector<int>& prev, int src,
145 int dst) -> std::vector<int> {
146 if (src == dst) return std::vector<int>{src};
147 if (dst < 0 || static_cast<size_t>(dst) >= prev.size() || prev[dst] == -1) {
148 return {};
149 }
150 std::vector<int> path;
151 int v = dst;
152 while (v != -1 && v != src) {
153 path.push_back(v);
154 v = prev[v];
155 }
156 if (v != src) return {};
157 path.push_back(src);
158 std::reverse(path.begin(), path.end());
159 return path;
160 }

238 template <typename Graph, typename WeightFunc>
239 auto solve_hadlock_component(const Graph& G, WeightFunc weight,
240 const std::vector<std::vector<typename Graph::node_t>>& faces)
241 -> py::set<std::pair<typename Graph::node_t, typename Graph::node_t>> {
242 using node_t = typename Graph::node_t;
243
244 // (1) Build the dual graph
245 const auto dual = build_dual<node_t>(faces, weight);
246
247 // (2) Identify odd-degree faces
248 std::vector<int> odd_faces;
249 for (size_t i = 0; i < faces.size(); ++i) {
250 if (faces[i].size() % 2 == 1) {
251 odd_faces.push_back(static_cast<int>(i));
252 }
253 }
254
255 // No odd faces -> graph is bipartite -> all edges are in the cut
256 if (odd_faces.size() < 2) {
257 py::set<std::pair<node_t, node_t>> all_edges;
258 for (const auto& e : G.edges()) {
259 all_edges.insert(e);
260 }
261 return all_edges;
262 }
263
264 // Handshaking on the face list (sum of face lengths = 2|E|) guarantees an
265 // even number of odd faces. An odd count means the supplied face set is
266 // malformed; dropping a terminal here would silently return a suboptimal cut.
267 if (odd_faces.size() % 2 != 0) {
268 throw xnetwork::XNetworkAlgorithmError(
269 "hadlock: odd number of odd-degree faces; the face list is malformed");
270 }
271
272 const auto n_odd = odd_faces.size();
273 const auto n_face = faces.size();
274
275 std::vector<char> is_odd(n_face, 0);
276 for (const auto f : odd_faces) is_odd[f] = 1;
277
278 // (3) Shortest paths from every odd face, keyed by face id (the matching
279 // weight function indexes rows by face id, not by odd-face position).
280 // Only predecessors are retained; matched-pair paths are reconstructed
281 // on demand in step (5), avoiding the O(n_odd^2) all-pairs path table.
282 std::vector<std::vector<int>> dist_mat(n_face);
283 std::vector<std::vector<int>> prev_mat(n_face);
284
285 for (size_t i = 0; i < n_odd; ++i) {
286 auto [dist, prev] = dijkstra<node_t>(dual, odd_faces[i]);
287 dist_mat[odd_faces[i]] = std::move(dist);
288 prev_mat[odd_faces[i]] = std::move(prev);
289 }
290
291 // (4) Minimum weight perfect matching on odd-face distances
292 auto matching = min_weight_perfect_matching<node_t>(odd_faces, dist_mat);
293
294 // (5) Collect primal edges excluded from the cut
295 // Build dual-edge -> primal-edge lookup
296 std::unordered_map<std::pair<int, int>, std::pair<node_t, node_t>> dedge_primal;
297 dedge_primal.reserve(dual.size() * 3);
298 for (size_t fi = 0; fi < dual.size(); ++fi) {
299 for (const auto& e : dual[fi]) {
300 int a = static_cast<int>(fi);
301 int b = e.neighbor;
302 if (a > b) std::swap(a, b);
303 dedge_primal[{a, b}] = e.primal;
304 }
305 }
306
307 // Walk each matched-pair path in the dual to find primal edges to exclude
308 std::set<std::pair<node_t, node_t>> excluded;
309 for (const auto& [u_face, v_face] : matching) {
310 if (!is_odd[u_face] || !is_odd[v_face]) continue;
311
312 const auto path = reconstruct_path(prev_mat[u_face], u_face, v_face);
313 for (size_t k = 0; k + 1 < path.size(); ++k) {
314 int a = path[k];
315 int b = path[k + 1];
316 if (a > b) std::swap(a, b);
317 auto it = dedge_primal.find({a, b});
318 if (it != dedge_primal.end()) {
319 auto [pu, pv] = it->second;
320 if (pu > pv) std::swap(pu, pv);
321 excluded.insert({pu, pv});
322 }
323 }
324 }
325
326 // (6) Max-cut = all edges not excluded
327 py::set<std::pair<node_t, node_t>> cut_edges;
328 for (const auto& e : G.edges()) {
329 auto [u, v] = e;
330 if (u > v) std::swap(u, v);
331 if (!excluded.count({u, v})) {
332 cut_edges.insert(e);
333 }
334 }
335 return cut_edges;
336 }

377template <typename Graph, typename WeightFunc> auto solve_hadlock_max_cut(
378 const Graph& G, WeightFunc weight,
379 const std::vector<std::vector<std::vector<typename Graph::node_t>>>& component_faces)
380 -> py::set<std::pair<typename Graph::node_t, typename Graph::node_t>> {
381 using node_t = typename Graph::node_t;
382 using edge_t = std::pair<node_t, node_t>;
383
384 // Decompose into biconnected components
385 auto comp_nodes = detail::biconnected_components(G);
386
387 if (comp_nodes.size() != component_faces.size()) {
388 throw xnetwork::XNetworkAlgorithmError(
389 "solve_hadlock_max_cut: component_faces count does not match the number of "
390 "biconnected components");
391 }
392 const auto n_comps = comp_nodes.size();
393
394 // Components are independent (edges are disjoint), so no
395 // synchronization is needed beyond collecting results.
396 xnetwork::thread_pool pool;
397 std::vector<std::future<py::set<edge_t>>> futures;
398 futures.reserve(n_comps);
399
400 for (size_t i = 0; i < n_comps; ++i) {
401 futures.push_back(pool.enqueue([&, i]() -> py::set<edge_t> {
402 const auto& comp_faces = component_faces[i];
403
404 auto comp_weight = [&weight](node_t u, node_t v) -> int { return weight(u, v); };
405
406 // Extract edges belonging to this component by scanning faces
407 std::vector<std::pair<node_t, node_t>> comp_edges;
408 for (const auto& face : comp_faces) {
409 for (size_t j = 0; j < face.size(); ++j) {
410 auto u = face[j];
411 auto v = face[(j + 1) % face.size()];
412 if (u > v) std::swap(u, v);
413 edge_t e{u, v};
414 if (std::find(comp_edges.begin(), comp_edges.end(), e) == comp_edges.end()) {
415 comp_edges.push_back(e);
416 }
417 }
418 }

443template <typename Graph, typename WeightFunc> auto validate_max_cut(
444 [[maybe_unused]] const Graph& G,
445 const py::set<std::pair<typename Graph::node_t, typename Graph::node_t>>& cut_edges,
446 WeightFunc weight) -> std::pair<bool, int> {
447 using node_t = typename Graph::node_t;
448
449 std::map<node_t, std::vector<node_t>> cut_adj;
450 int total_weight = 0;
451 for (const auto& e : cut_edges) {
452 cut_adj[e.first].push_back(e.second);
453 cut_adj[e.second].push_back(e.first);
454 total_weight += weight(e.first, e.second);
455 }
456
457 std::map<node_t, int> colour;
458 bool is_bipartite = true;
459 for (const auto& [start, _] : cut_adj) {
460 if (colour.count(start)) continue;
461 colour[start] = 1;
462 std::queue<node_t> q;
463 q.push(start);
464 while (!q.empty() && is_bipartite) {
465 auto u = q.front();
466 q.pop();
467 for (const auto& v : cut_adj[u]) {
468 if (!colour.count(v)) {
469 colour[v] = (colour[u] == 1) ? 2 : 1;
470 q.push(v);
471 } else if (colour[v] == colour[u]) {
472 is_bipartite = false;
473 break;
474 }
475 }
476 }
477 }
478 return {is_bipartite, total_weight};
479}
