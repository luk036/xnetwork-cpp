74 template <typename Node, typename WeightFunc> auto prim_mst(size_t n, WeightFunc&& weight)
75 -> std::vector<std::pair<Node, Node>> {
76 constexpr double INF = std::numeric_limits<double>::max();
77
78 std::vector<bool> in_mst(n, false);
79 std::vector<double> key(n, INF);
80 std::vector<Node> parent(n, Node{0});
81
82 key[0] = 0.0;
83
84 for (size_t count = 0; count < n; ++count) {
85 // Find the vertex with smallest key not yet in MST
86 double best = INF;
87 Node u = Node{0};
88 for (Node v = Node{0}; v < static_cast<Node>(n); ++v) {
89 if (!in_mst[static_cast<size_t>(v)] && key[static_cast<size_t>(v)] < best) {
90 best = key[static_cast<size_t>(v)];
91 u = v;
92 }
93 }
94 if (best == INF) break; // isolated vertex (shouldn't happen for complete graphs)
95
96 in_mst[static_cast<size_t>(u)] = true;
97
98 // Relaxation
99 for (Node v = Node{0}; v < static_cast<Node>(n); ++v) {
100 if (!in_mst[static_cast<size_t>(v)]) {
101 const double w = std::forward<WeightFunc>(weight)(u, v);
102 auto& kv = key[static_cast<size_t>(v)];
103 if (w < kv) {
104 kv = w;
105 parent[static_cast<size_t>(v)] = u;
106 }
107 }
108 }
109 }
110
111 std::vector<std::pair<Node, Node>> edges;
112 edges.reserve(n - 1);
113 for (Node v = Node{1}; v < static_cast<Node>(n); ++v) {
114 const auto p = parent[static_cast<size_t>(v)];
115 if (p != v) // safety
116 edges.emplace_back(p, v);
117 }
118 return edges;
119 }

124 template <typename Node>
125 auto find_odd_degree_nodes(const std::vector<std::pair<Node, Node>>& mst_edges, size_t n)
126 -> std::vector<Node> {
127 std::vector<int> deg(n, 0);
128 for (const auto& [u, v] : mst_edges) {
129 ++deg[static_cast<size_t>(u)];
130 ++deg[static_cast<size_t>(v)];
131 }
132 std::vector<Node> odd;
133 for (Node v = Node{0}; v < static_cast<Node>(n); ++v) {
134 if (deg[static_cast<size_t>(v)] % 2 != 0) odd.push_back(v);
135 }
136 return odd;
137 }

177 -> std::vector<Node> {

179 py::set<Node> visited;

219template <typename Graph, typename WeightFunc>
220auto christofides_tsp(const Graph& G, WeightFunc&& weight) -> std::vector<typename Graph::node_t> {
221 using Node = typename Graph::node_t;
222 const size_t n = G.number_of_nodes();
223
224 if (n == 0) return {};
225 if (n == 1) return {Node{0}, Node{0}};
226
227 // 1. Minimum Spanning Tree
228 const auto mst_edges = detail::prim_mst<Node>(n, weight);
229
230 // 2. Odd-degree vertices in the MST
231 const auto odd_nodes = detail::find_odd_degree_nodes(mst_edges, n);
232
233 // 3. Minimum-weight perfect matching on odd vertices (Edmonds' blossom)
234 const auto matching
235 = detail::blossom_min_weight_perfect_matching(odd_nodes, std::forward<WeightFunc>(weight));
236
237 // 4. Build the Eulerian multigraph (MST + matching)
238 auto adj = detail::build_multigraph<Node>(n, mst_edges, matching);
239
240 // 5. Eulerian circuit via Hierholzer
241 const auto eulerian_circuit = detail::hierholzer<Node>(std::move(adj), Node{0});
242
243 // 6. Shortcut -> Hamiltonian cycle
244 return detail::shortcut_eulerian(eulerian_circuit);
245}

293template <typename Graph, typename WeightFunc>
294auto two_opt(std::vector<typename Graph::node_t> path, const Graph& G, WeightFunc&& weight)
295 -> std::vector<typename Graph::node_t> {
296 (void)G; // only used for type deduction
297
298 bool improved = true;
299 while (improved) {
300 improved = false;
301 for (size_t i = 1; i + 2 < path.size(); ++i) {
302 for (size_t j = i + 1; j + 1 < path.size(); ++j) {
303 if (j - i == 1) continue;
304
305 const double delta = -weight(path[i - 1], path[i]) - weight(path[j], path[j + 1])
306 + weight(path[i - 1], path[j])
307 + std::forward<WeightFunc>(weight)(path[i], path[j + 1]);
308
309 if (delta < -1.0e-12) {
310 std::reverse(path.begin() + static_cast<ptrdiff_t>(i),
311 path.begin() + static_cast<ptrdiff_t>(j + 1));
312 improved = true;
313 }
314 }
315 }
316 }
317 return path;
318}
