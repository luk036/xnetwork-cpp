51template <typename MakeViolator, typename WeightMap, typename SolutionSet>
52auto pd_cover(MakeViolator make_violator, WeightMap& weight,
53 SolutionSet& soln) -> std::pair<SolutionSet, typename WeightMap::mapped_type> {
54 using CostType = typename WeightMap::mapped_type;
55 using NodeType = typename SolutionSet::value_type;
56
57 CostType total_dual_cost = 0;
58 auto gap = weight; // copy weights
59 std::vector<NodeType> added_order;
60
61 // Phase 1: Primal-Dual Selection
62 // Repeatedly call the violator for each violation, updating
63 // coverset/gap between calls (lazy evaluation equivalent to the
64 // original coroutine-based generator).
65 {
66 auto next = make_violator();
67 while (auto opt = next()) {
68 auto& violate_set = *opt;
69 if (violate_set.empty()) continue;
70
71 auto min_vtx = *std::min_element(
72 violate_set.begin(), violate_set.end(),
73 [&](const auto& v1, const auto& v2) { return gap[v1] < gap[v2]; });
74 auto min_val = gap[min_vtx];
75
76 if (!soln.contains(min_vtx)) {
77 soln.insert(min_vtx);
78 added_order.emplace_back(min_vtx);
79 }
80
81 total_dual_cost += min_val;
82
83 for (const auto& vtx : violate_set) {
84 gap[vtx] -= min_val;
85 }
86 }
87 }
88
89 // Phase 2: Reverse-Delete Post-Processing
90 for (auto it = added_order.rbegin(); it != added_order.rend(); ++it) {
91 soln.erase(*it);
92 bool is_redundant = true;
93 {
94 auto check = make_violator();
95 while (auto opt = check()) {
96 if (!opt->empty()) {
97 is_redundant = false;
98 break;
99 }
100 }
101 }
102 if (!is_redundant) {
103 soln.insert(*it);
104 }
105 }
106
107 CostType final_prml_cost = 0;
108 for (const auto& vtx : soln) {
109 final_prml_cost += weight[vtx];
110 }
111
112 if (total_dual_cost > final_prml_cost) {
113 throw std::logic_error(
114 "pd_cover: primal-dual invariant violated (dual cost exceeds primal cost)");
115 }
116 return std::make_pair(soln, final_prml_cost);
117}

229template <typename Graph, typename WeightMap, typename CoverSet>
230auto min_cycle_cover(const Graph& ugraph, WeightMap& weight,
231 CoverSet& coverset) -> std::pair<CoverSet, typename WeightMap::mapped_type> {
232 using node_t = typename Graph::node_t;
233
234 // Factory: returns a violator that does a fresh BFS each call
235 // and returns the first cycle found (or nullopt if none).
236 auto make_violate = [&]() {
237 return [&ugraph, &coverset]() -> std::optional<std::vector<node_t>> {
238 auto cycles = generic_bfs_cycle<Graph, CoverSet>(ugraph, coverset);
239 if (cycles.empty()) return std::nullopt;
240 const auto& [info, parent, child] = cycles[0];
241 auto cycle_deque = construct_cycle<node_t>(info, parent, child);
242 return std::vector<node_t>(cycle_deque.begin(), cycle_deque.end());
243 };
244 };
245
246 return pd_cover(make_violate, weight, coverset);
247}
