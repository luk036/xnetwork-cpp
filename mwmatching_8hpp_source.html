103template <typename WeightType>
104void check_input_graph(const std::vector<Edge<WeightType>>& edges)
105{
106 const VertexId max_num_vertex = std::numeric_limits<VertexId>::max();
107
108 if (edges.size() >= NO_EDGE) {
109 throw std::invalid_argument("Too many edges");
110 }
111
112 for (const Edge<WeightType>& edge : edges) {
113
114 // Check that vertex IDs are valid.
115 if ((edge.vt.first >= max_num_vertex)
116 || (edge.vt.second >= max_num_vertex)) {
117 throw std::invalid_argument("Vertex ID out of range");
118 }
119
120 // Check that edge weight is a finite number.
121 if (! std::numeric_limits<WeightType>::is_integer) {
122 if (! std::isfinite(edge.weight)) {
123 throw std::invalid_argument(
124 "Edge weights must be finite numbers");
125 }
126 }
127
128 // Check that edge weight will not cause overflow.
129 if (edge.weight > std::numeric_limits<WeightType>::max() / 6) {
130 throw std::invalid_argument(
131 "Edge weight exceeds maximum supported value");
132 }
133 }
134
135 // Check that the graph has no self-edges.
136 for (const Edge<WeightType>& edge : edges) {
137 if (edge.vt.first == edge.vt.second) {
138 throw std::invalid_argument("Self-edges are not supported");
139 }
140 }
141
142 // Check that the graph does not contain duplicate edges.
143 std::vector<VertexPair> edge_endpoints;
144 edge_endpoints.reserve(edges.size());
145 for (const Edge<WeightType>& edge : edges) {
146 VertexPair vt = edge.vt;
147 if (vt.first > vt.second) {
148 std::swap(vt.first, vt.second);
149 }
150 edge_endpoints.push_back(vt);
151 }
152
153 std::sort(edge_endpoints.begin(), edge_endpoints.end());
154 if (std::unique(edge_endpoints.begin(), edge_endpoints.end())
155 != edge_endpoints.end()) {
156 throw std::invalid_argument("Duplicate edges are not supported");
157 }
158}

246 static std::vector<std::vector<EdgeId>> build_adjacent_edges(
247 const std::vector<EdgeT>& edges,
248 VertexId num_vertex)
249 {
250 // Count the number of incident edges per vertex.
251 std::vector<VertexId> vertex_degree(num_vertex);
252 for (const EdgeT& edge : edges) {
253 assert(edge.vt.first < num_vertex);
254 assert(edge.vt.second < num_vertex);
255 vertex_degree[edge.vt.first]++;
256 vertex_degree[edge.vt.second]++;
257 }
258
259 // Build adjacency lists.
260 std::vector<std::vector<EdgeId>> adjacent_edges(num_vertex);
261 for (VertexId i = 0; i < num_vertex; ++i) {
262 adjacent_edges[i].reserve(vertex_degree[i]);
263 }
264 for (EdgeId e = 0; e < edges.size(); e++) {
265 const EdgeT& edge = edges[e];
266 adjacent_edges[edge.vt.first].push_back(e);
267 adjacent_edges[edge.vt.second].push_back(e);
268 }

479template <typename WeightType, typename Func>
480inline void for_vertices_in_blossom(const Blossom<WeightType>* blossom, Func func)
481{
482 const NonTrivialBlossom<WeightType>* ntb = blossom->nontrivial();
483 if (ntb) {
484 // Visit all vertices in the non-trivial blossom.
485 // Use an explicit stack to avoid deep call chains.
486 std::vector<const NonTrivialBlossom<WeightType>*> stack;
487 stack.push_back(ntb);
488
489 while (! stack.empty()) {
490 const NonTrivialBlossom<WeightType>* b = stack.back();
491 stack.pop_back();
492
493 for (const auto& sub : b->subblossoms) {
494 ntb = sub.blossom->nontrivial();
495 if (ntb) {
496 stack.push_back(ntb);
497 } else {
498 func(sub.blossom->base_vertex);
499 }
500 }
501 }
502
503 } else {
504 // A trivial blossom contains just one vertex.
505 func(blossom->base_vertex);
506 }

751 void delta2_add_edge(EdgeId e, VertexId y, BlossomT* by)
752 {
753 WeightType prio = edge_pseudo_slack(e);
754
755 // Insert the edge in the S-edge queue of vertex "y".
756 // Check whether this improves the min-prio of the vertex.
757 assert(! vertex_sedge_node[e].valid());
758 bool improved = vertex_sedge_queue[y].empty()
759 || (prio < vertex_sedge_queue[y].min_prio());
760 vertex_sedge_queue[y].insert(&vertex_sedge_node[e], prio, e);
761
762 if (improved) {
763 // Update the priority of vertex "y" in its ConcatenableQueue.
764 vertex_queue_node[y].set_prio(prio);
765
766 // Update the global delta2 queue if this blossom is unlabeled
767 // and the new edge becomes its least-slack S-edge.
768 if (by->label == LABEL_NONE) {
769 prio += by->vertex_dual_offset;
770 if (by->delta2_node.valid()) {
771 if (prio < by->delta2_node.prio()) {
772 delta2_queue.set_prio(&by->delta2_node, prio);
773 }
774 } else {
775 delta2_queue.insert(&by->delta2_node, prio, by);
776 }
777 }
778 }

1102

1106 // S-vertex "y". It must be tracked for delta2.

1110 // T-vertex or unlabeled vertex "y". This edge was
1111 // tracked for delta2 but must now be removed.
1112 delta2_remove_edge(e, y, by);
1113 }
1114 }
1115 });

1223 AlternatingPath trace_alternating_paths(VertexId x, VertexId y)
1224 {
1225 // Initialize a path containing only the edge (x, y).
1226 AlternatingPath path;
1227 path.edges.emplace_back(x, y);
1228
1229 // "first_common" is the first common ancestor of "x" and "y"
1230 // in the alternating tree, or "nullptr" if no common ancestor
1231 // has been found.
1232 BlossomT* first_common = nullptr;
1233
1234 // Alternate between tracing the path from "x" and the path from "y".
1235 // This ensures that the search time is bounded by the size of any
1236 // newly found blossom.
1237 while ((x != NO_VERTEX) || (y != NO_VERTEX)) {
1238
1239 // Trace path from vertex "x".
1240 if (x != NO_VERTEX) {
1241
1242 // Stop if we found a common ancestor.
1243 BlossomT* bx = top_level_blossom(x);
1244 if (vertex_marker[bx->base_vertex]) {
1245 first_common = bx;
1246 break;
1247 }
1248
1249 // Mark blossom as potential common ancestor.
1250 vertex_marker[bx->base_vertex] = true;
1251 marked_vertex.push_back(bx->base_vertex);
1252
1253 // Trace back to the parent in the alternating tree.
1254 x = bx->tree_edge.first;
1255 if (x != NO_VERTEX) {
1256 path.edges.push_front(bx->tree_edge);
1257 }
1258 }
1259
1260 // Trace path from vertex "y".
1261 if (y != NO_VERTEX) {
1262
1263 // Stop if we found a common ancestor.
1264 BlossomT* by = top_level_blossom(y);
1265 if (vertex_marker[by->base_vertex]) {
1266 first_common = by;
1267 break;
1268 }
1269
1270 // Mark blossom as potential common ancestor.
1271 vertex_marker[by->base_vertex] = true;
1272 marked_vertex.push_back(by->base_vertex);
1273
1274 // Trace back to the parent in the alternating tree.
1275 y = by->tree_edge.first;
1276 if (y != NO_VERTEX) {
1277 path.edges.emplace_back(by->tree_edge.second, y);
1278 }
1279 }
1280 }
1281
1282 // Remove all markers we placed.
1283 for (VertexId k : marked_vertex) {
1284 vertex_marker[k] = false;
1285 }
1286 marked_vertex.clear();
1287
1288 // If we found a common ancestor, trim the paths so they end there.
1289 if (first_common) {
1290 assert(first_common->label == LABEL_S);
1291 while (top_level_blossom(path.edges.front().first)
1292 != first_common) {
1293 path.edges.pop_front();
1294 }
1295 while (top_level_blossom(path.edges.back().second)
1296 != first_common) {
1297 path.edges.pop_back();
1298 }
1299 }
1300
1301 // Any alternating path between S-blossoms must have odd length.
1302 assert(path.edges.size() % 2 == 1);
1303
1304 return path;
1305 }

1315 void make_blossom(const AlternatingPath& path)
1316 {
1317 assert(path.edges.size() % 2 == 1);
1318 assert(path.edges.size() >= 3);
1319
1320 // Collect pointers to sub-blossoms.
1321 std::vector<BlossomT*> subblossoms;
1322 subblossoms.reserve(path.edges.size());
1323 for (VertexPair edge : path.edges) {
1324 subblossoms.push_back(top_level_blossom(edge.first));
1325 }
1326
1327 // Check that the path is cyclic.
1328 for (std::size_t i = 0; i < path.edges.size(); ++i) {
1329 assert(top_level_blossom(path.edges[i].second)
1330 == subblossoms[(i + 1) % subblossoms.size()]);
1331 }
1332
1333 // Blossom must start and end with an S-blossom.
1334 assert(subblossoms.front()->label == LABEL_S);
1335
1336 for (BlossomT* sub : subblossoms) {
1337
1338 // Mark vertices inside former T-blossoms as S-vertices.
1339 if (sub->label == LABEL_T) {
1340 remove_blossom_label_t(sub);
1341 assign_blossom_label_s(sub);
1342 }
1343
1344 // Remove labels from sub-blossoms.
1345 change_s_blossom_to_subblossom(sub);
1346 }
1347
1348 // Create the new blossom object.
1349 auto blossom_iterator = nontrivial_blossom.emplace(
1350 nontrivial_blossom.end(),
1351 subblossoms,
1352 path.edges);
1353 NonTrivialBlossomT* blossom = &(*blossom_iterator);
1354 blossom->this_blossom_iterator = blossom_iterator;
1355
1356 // Initialize modified blossom dual to set true dual to 0.
1357 blossom->dual_var = -2 * delta_sum;
1358
1359 // Link the subblossoms to the their new parent.
1360 for (BlossomT* sub : subblossoms) {
1361 sub->parent = blossom;
1362 }
1363
1364 // Merge concatenable queues.
1365 std::vector<VertexQueueT*> subqueues;
1366 subqueues.reserve(subblossoms.size());
1367 for (BlossomT* sub : subblossoms) {
1368 subqueues.push_back(&sub->vertex_queue);
1369 }
1370 blossom->vertex_queue.merge(subqueues.begin(), subqueues.end());
1371
1372 // Assign label S to the new blossom and link to the alternating tree.
1373 blossom->label = LABEL_S;
1374 blossom->tree_edge = subblossoms.front()->tree_edge;
1375 blossom->tree_root = subblossoms.front()->tree_root;
1376 }

1379 void expand_t_blossom(NonTrivialBlossomT* blossom)
1380 {
1381 assert(! blossom->parent);
1382 assert(blossom->label == LABEL_T);
1383
1384 // Remove label from blossom.
1385 remove_blossom_label_t(blossom);
1386
1387 // Remove blossom from the delta2 queue.
1388 delta2_disable_blossom(blossom);
1389
1390 // Split concatenable queue, thus reconstructing the separate
1391 // concatenable queues of the sub-blossoms.
1392 blossom->vertex_queue.split();
1393
1394 // Prepare to push pending delta updates down to the sub-blossoms.
1395 WeightType vertex_dual_offset = blossom->vertex_dual_offset;
1396 blossom->vertex_dual_offset = 0;
1397
1398 // Convert sub-blossoms into top-level blossoms.
1399 for (const auto& sub : blossom->subblossoms) {
1400 BlossomT* sub_blossom = sub.blossom;
1401 assert(sub_blossom->parent == blossom);
1402 assert(sub_blossom->label == LABEL_NONE);
1403 sub_blossom->parent = nullptr;
1404
1405 // Push pending delta updates to sub-blossom.
1406 assert(sub_blossom->vertex_dual_offset == 0);
1407 sub_blossom->vertex_dual_offset = vertex_dual_offset;
1408
1409 // Add unlabeled sub-blossom to the delta2 queue.
1410 delta2_enable_blossom(sub_blossom);
1411 }
1412
1413 // The expanded blossom was part of an alternating tree.
1414 // We must now reconstruct the part of the alternating tree
1415 // that ran through this blossom by linking some of the sub-blossoms
1416 // into the tree.
1417
1418 // Find the sub-blossom through which the expanding blossom was
1419 // attached to the alternating tree.
1420 BlossomT* entry = top_level_blossom(blossom->tree_edge.second);
1421
1422 // Find the position of this sub-blossom within the expanding blossom.
1423 auto subblossom_loc = blossom->find_subblossom(entry);
1424 VertexId entry_pos = subblossom_loc.first;
1425 auto entry_it = subblossom_loc.second;
1426
1427 // Get the edge that attaches this blossom to the alternating tree.
1428 VertexId x, y;
1429 std::tie(x, y) = blossom->tree_edge;
1430
1431 // Walk around the expanding blossom from "entry" to its base
1432 // in an even number of steps.
1433 auto sub_begin = blossom->subblossoms.begin();
1434 auto sub_end = blossom->subblossoms.end();
1435 auto sub_it = entry_it;
1436 while ((sub_it != sub_begin) && (sub_it != sub_end)) {
1437
1438 // Assign label T to the current sub-blossom on the path.
1439 // This also assigns label S to the next sub-blossom.
1440 extend_tree_s_to_t(x, y);
1441
1442 if (entry_pos % 2 == 0) {
1443 // Walk two steps backward to the base.
1444 --sub_it;
1445 assert(sub_it != sub_begin);
1446 --sub_it;
1447 // Get the edge from S-sub-blossom to next sub-blossom.
1448 std::tie(y, x) = sub_it->edge;
1449 } else {
1450 // Walk two steps forward to the base.
1451 ++sub_it;
1452 assert(sub_it != sub_end);
1453 // Get the edge from S-sub-blossom to next sub-blossom.
1454 std::tie(x, y) = sub_it->edge;
1455 ++sub_it;
1456 }
1457 }
1458
1459 // Finally, assign label T to the base sub-blossom.
1460 BlossomT* base = blossom->subblossoms.front().blossom;
1461 assign_blossom_label_t(base);
1462 base->tree_edge = std::make_pair(x, y);
1463 base->tree_root = blossom->tree_root;
1464
1465 // Delete the expanded blossom.
1466 nontrivial_blossom.erase(blossom->this_blossom_iterator);
1467 }

1478 void augment_blossom_rec(
1479 NonTrivialBlossomT* blossom,
1480 BlossomT* entry,
1481 std::stack<std::pair<NonTrivialBlossomT*, BlossomT*>>& rec_stack)
1482 {
1483 // Find the position of "entry" within "blossom".
1484 auto subblossom_loc = blossom->find_subblossom(entry);
1485 VertexId entry_pos = subblossom_loc.first;
1486 auto entry_it = subblossom_loc.second;
1487
1488 // Walk around the blossom from "entry" to the base
1489 // in an even number of steps.
1490 auto sub_begin = blossom->subblossoms.begin();
1491 auto sub_end = blossom->subblossoms.end();
1492 auto sub_it = entry_it;
1493 while ((sub_it != sub_begin) && (sub_it != sub_end)) {
1494 VertexId x, y;
1495 BlossomT* bx;
1496 BlossomT* by;
1497
1498 if (entry_pos % 2 == 0) {
1499 // Walk backward to the base.
1500 --sub_it;
1501 by = sub_it->blossom;
1502 assert(sub_it != sub_begin);
1503 --sub_it;
1504 bx = sub_it->blossom;
1505 std::tie(x, y) = sub_it->edge;
1506 } else {
1507 // Walk forward to the base.
1508 // We may wrap past the end of the subblossom list.
1509 ++sub_it;
1510 assert(sub_it != sub_end);
1511 std::tie(x, y) = sub_it->edge;
1512 bx = sub_it->blossom;
1513 ++sub_it;
1514 by = (sub_it == sub_end) ?
1515 blossom->subblossoms.front().blossom :
1516 sub_it->blossom;
1517 }
1518
1519 // Pull this edge into the matching.
1520 vertex_mate[x] = y;
1521 vertex_mate[y] = x;
1522
1523 // Augment through any non-trivial subblossoms touching this edge.
1524 NonTrivialBlossomT* bx_ntb = bx->nontrivial();
1525 if (bx_ntb) {
1526 rec_stack.emplace(bx_ntb, &trivial_blossom[x]);
1527 }
1528
1529 NonTrivialBlossomT* by_ntb = by->nontrivial();
1530 if (by_ntb) {
1531 rec_stack.emplace(by_ntb, &trivial_blossom[y]);
1532 }
1533 }
1534
1535 // Re-orient the blossom.
1536 if (entry_it != sub_begin) {
1537 // Rotate the subblossom list.
1538 blossom->subblossoms.splice(sub_begin,
1539 blossom->subblossoms,
1540 entry_it,
1541 sub_end);
1542 }
1543
1544 // Update the base vertex.
1545 // We can pull the new base vertex from the entry sub-blossom
1546 // since its augmentation has already finished.
1547 blossom->base_vertex = entry->base_vertex;
1548 }
1549
1556 void augment_blossom(NonTrivialBlossomT* blossom, BlossomT* entry)
1557 {
1558 // Use an explicit stack to avoid deep recursion.
1559 std::stack<std::pair<NonTrivialBlossomT*, BlossomT*>> rec_stack;
1560 rec_stack.emplace(blossom, entry);
1561
1562 while (!rec_stack.empty()) {
1563 NonTrivialBlossomT* outer_blossom;
1564 BlossomT* inner_entry;
1565 std::tie(outer_blossom, inner_entry) = rec_stack.top();
1566
1567 NonTrivialBlossomT* inner_blossom = inner_entry->parent;
1568 assert(inner_blossom);
1569
1570 if (inner_blossom != outer_blossom) {
1571 // After augmenting "inner_blossom",
1572 // continue by augmenting its parent.
1573 rec_stack.top() = std::make_pair(outer_blossom,
1574 inner_blossom);
1575 } else {
1576 // After augmenting "inner_blossom",
1577 // this entire "outer_blossom" will be finished.
1578 rec_stack.pop();
1579 }
1580
1581 // Augment "inner_blossom".
1582 augment_blossom_rec(inner_blossom, inner_entry, rec_stack);
1583 }

1599 top_level_blossom(path.edges.back().second)->base_vertex]
1600 == NO_VERTEX);
1601

1603 auto edge_it = path.edges.begin();

1775 /* ********** Delta steps: ********** */
1776
1795 DeltaStep calc_dual_delta_step()
1796 {
1797 DeltaStep delta;
1798 delta.blossom = nullptr;
1799
1800 // Compute delta1: minimum dual variable of any S-vertex.
1801 // All unmatched vertices have the same dual value, and this is
1802 // the minimum value among all S-vertices.
1803 delta.kind = 1;
1804 delta.value = init_vertex_dual - delta_sum;
1805
1806 // Compute delta2: minimum slack of any edge between an S-vertex and
1807 // an unlabeled vertex.
1808 EdgeId e;
1809 WeightType slack;
1810 std::tie(e, slack) = delta2_get_min_edge();
1811 if ((e != NO_EDGE) && (slack <= delta.value)) {
1812 delta.kind = 2;
1813 delta.value = slack;
1814 delta.edge = graph.edges[e].vt;
1815 }
1816
1817 // Compute delta3: half minimum slack of any edge between two
1818 // top-level S-blossoms.
1819 std::tie(e, slack) = delta3_get_min_edge();
1820 if ((e != NO_EDGE) && (slack / 2 <= delta.value)) {
1821 delta.kind = 3;
1822 delta.value = slack / 2;
1823 delta.edge = graph.edges[e].vt;
1824 }
1825
1826 // Compute delta4: half minimum dual of a top-level T-blossom.
1827 if (! delta4_queue.empty()) {
1828 NonTrivialBlossomT* blossom = delta4_queue.min_elem();
1829 assert(! blossom->parent);
1830 assert(blossom->label == LABEL_T);
1831 // Calculate true blossom dual variable from modified dual.
1832 WeightType blossom_dual = blossom->dual_var - 2 * delta_sum;
1833 if (blossom_dual / 2 <= delta.value) {
1834 delta.kind = 4;
1835 delta.value = blossom_dual / 2;
1836 delta.blossom = blossom;
1837 }
1838 }
1839
1840 return delta;
1841 }

1881 bool run_stage()
1882 {
1883 // Each pass through the following loop is a "substage".
1884 // The substage tries to find an augmenting path.
1885 // If an augmenting path is found, we augment the matching and end
1886 // the stage. Otherwise we update the dual LPP problem and enter the
1887 // next substage, or stop if no further improvement is possible.
1888 //
1889 // This loop runs through at most O(n) iterations per stage.
1890 while (true) {
1891
1892 // Consider the incident edges of newly labeled S-vertices.
1893 scan_new_s_vertices();
1894
1895 // Calculate delta step in the dual LPP problem.
1896 DeltaStep delta = calc_dual_delta_step();
1897
1898 // Update the running sum of delta steps. This implicitly updates
1899 // the dual variables of vertices and blossoms.
1900 delta_sum += delta.value;
1901
1902 if (delta.kind == 2) {
1903 // Use the edge from S-vertex to unlabeled vertex that got
1904 // unlocked through the delta update.
1905 VertexId x = delta.edge.first;
1906 VertexId y = delta.edge.second;
1907 if (top_level_blossom(x)->label != LABEL_S) {
1908 std::swap(x, y);
1909 }
1910 extend_tree_s_to_t(x, y);
1911
1912 } else if (delta.kind == 3) {
1913 // Use the S-to-S edge that got unlocked by the delta update.
1914 // This may reveal an augmenting path.
1915 VertexId x = delta.edge.first;
1916 VertexId y = delta.edge.second;
1917 bool augmented = add_s_to_s_edge(x, y);
1918 if (augmented) {
1919 return true;
1920 }
1921
1922 } else if (delta.kind == 4) {
1923 // Expand the T-blossom that reached dual value 0 through
1924 // the delta update.
1925 assert(delta.blossom);
1926 expand_t_blossom(delta.blossom);
1927
1928 } else {
1929 // No further improvement possible. End the stage.
1930 assert(delta.kind == 1);
1931 return false;
1932 }
1933 }
1934 }
1935
1942 void cleanup()
1943 {
1944 assert(scan_queue.empty());
1945
1946 auto cleanup_blossom = [this](BlossomT* blossom) {
1947
1948 // Remove blossom label.
1949 if ((! blossom->parent) && (blossom->label != LABEL_NONE)) {
1950 reset_blossom_label(blossom);
1951 }
1952
1953 // Unwind lazy delta updates to vertex dual variables.
1954 if (blossom->vertex_dual_offset != 0) {
1955 WeightType dual_fixup = blossom->vertex_dual_offset;
1956 blossom->vertex_dual_offset = 0;
1957 for_vertices_in_blossom(blossom,
1958 [this,dual_fixup](VertexId x) {
1959 vertex_dual[x] += dual_fixup;
1960 });
1961 }
1962 };
1963
1964 for (BlossomT& blossom : trivial_blossom) {
1965 cleanup_blossom(&blossom);
1966 }
1967 for (BlossomT& blossom : nontrivial_blossom) {
1968 cleanup_blossom(&blossom);
1969 }
1970
1971 assert(delta2_queue.empty());
1972 assert(delta3_queue.empty());
1973 assert(delta4_queue.empty());
1974 }
1975
1977 void run()
1978 {
1979 // Assign label S to all vertices.
1980 start();
1981
1982 // Improve the solution until no further improvement is possible.
1983 //
1984 // Each successful pass through this loop increases the number
1985 // of matched edges by 1.
1986 //
1987 // This loop runs through at most (n/2 + 1) iterations.
1988 // Each iteration takes time O((n + m) * log(n)).
1989 while (run_stage()) ;
1990
1991 // Clean up and unwind lazy updates to dual variables.
1992 cleanup();
1993 }

2427template <typename WeightType>
2428std::vector<Edge<WeightType>> adjust_weights_for_maximum_cardinality_matching(
2429 const std::vector<Edge<WeightType>>& edges_in)
2430{

2432 const WeightType max_safe_weight = std::numeric_limits<WeightType>::max() / 6;
2433
2434 // Copy edges.
2435 std::vector<Edge<WeightType>> edges(edges_in);
2436
2437 // Don't worry about empty graphs.
2438 if (edges.empty()) {
2439 return edges;
2440 }
2441
2442 // Count number of vertices.
2443 // Determine minimum and maximum edge weight.
2444 VertexId num_vertex = 0;
2445 WeightType min_weight = edges.front().weight;
2446 WeightType max_weight = min_weight;
2447
2448 const VertexId max_num_vertex = std::numeric_limits<VertexId>::max();
2449 for (const Edge<WeightType>& edge : edges) {
2450 VertexId m = std::max(edge.vt.first, edge.vt.second);
2451 if (m >= max_num_vertex) {
2452 throw std::invalid_argument("Vertex ID out of range");
2453 }
2454 num_vertex = std::max(num_vertex, m + 1);
2455
2456 if (! std::numeric_limits<WeightType>::is_integer) {
2457 if (! std::isfinite(edge.weight)) {
2458 throw std::invalid_argument(
2459 "Edge weights must be finite numbers");
2460 }
2461 }
2462
2463 min_weight = std::min(min_weight, edge.weight);
2464 max_weight = std::max(max_weight, edge.weight);
2465 }
2466
2467 // Calculate weight range and required weight adjustment.
2468 if ((min_weight < min_safe_weight) || (max_weight > max_safe_weight)) {
2469 throw std::invalid_argument(
2470 "Edge weight exceeds maximum supported value");
2471 }
2472
2473 WeightType weight_range = max_weight - min_weight;
2474
2475 if (weight_range > max_safe_weight / num_vertex) {
2476 throw std::invalid_argument(
2477 "Adjusted edge weight exceeds maximum supported value");
2478 }
2479
2480 // Do nothing if the weights already ensure maximum-cardinality.
2481 if ((min_weight > 0) && (min_weight >= num_vertex * weight_range)) {
2482 return edges;
2483 }
2484
2485 WeightType delta;
2486 if (weight_range > 0) {
2487 // Increase weights to make minimum edge weight large enough
2488 // to improve any non-maximum-cardinality matching.
2489 delta = num_vertex * weight_range - min_weight;
2490 } else {
2491 // All weights are the same. Increase weights to make them positive.
2492 delta = 1 - min_weight;
2493 }
2494
2495 assert(delta >= 0);
2496
2497 if (delta > max_safe_weight - max_weight) {
2498 throw std::invalid_argument(
2499 "Adjusted edge weight exceeds maximum supported value");
2500 }
2501

2503 for (Edge<WeightType>& edge: edges) {
2504 edge.weight += delta;
2505 }
2506
2507 return edges;
